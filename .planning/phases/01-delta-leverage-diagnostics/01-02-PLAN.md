```python
def test_page13_advanced_econometrics_delta_leverage_smoke(self, full_panel):
    """
    Phase 1 integration smoke: page 13 is the only UI consumer of all 4 Phase 1
    backend functions. This test verifies:
      1. The 3 functions page 13 imports from models.econometric still exist
         and are importable by their public + private names.
      2. _compute_delta_leverage (private, imported at page13:295-296) still works.
      3. run_delta_leverage_all returns the keys page 13 reads at lines 181+.
      4. run_delta_leverage_by_stage returns the keys page 13 iterates at lines 243+.
    Does NOT execute Streamlit UI — page 13 has st.* side effects at module
    scope. Tests the function-level contract page 13 depends on.
    """
    # 1. Imports page 13 declares (lines 17-19 of pages/13_advanced_econometrics.py)
    from models.econometric import (
        run_system_gmm,
        run_delta_leverage_all,
        run_delta_leverage_by_stage,
        run_stage_comparison,
        run_breusch_pagan_lm,
        run_pooled_ols,
    )
    # Private import page 13 makes inline at line 295-296
    from models.econometric import _compute_delta_leverage

    # 2. _compute_delta_leverage end-to-end (drops first obs per firm)
    delta_df = _compute_delta_leverage(full_panel)
    assert "delta_leverage" in delta_df.columns
    assert len(delta_df) > 0
    assert len(delta_df) < len(full_panel), \
        "First-difference must drop at least one row per firm"

    # 3. run_delta_leverage_all — keys consumed by page 13 lines 181+
    result = run_delta_leverage_all(full_panel)
    for required in ("ols", "fe", "re", "hausman", "recommended"):
        assert required in result, f"page13 contract: result['{required}']"
    for sub in ("ols", "fe", "re"):
        assert "coef_table" in result[sub]
        assert "r_squared" in result[sub]
        assert "n_obs" in result[sub]
    for hkey in ("chi2", "p_value", "verdict"):
        assert hkey in result["hausman"], f"page13 contract: hausman['{hkey}']"

    # 4. run_delta_leverage_by_stage — keys page 13 iterates at lines 243+
    by_stage = run_delta_leverage_by_stage(full_panel)
    assert isinstance(by_stage, dict) and len(by_stage) > 0
    has_real_result = False
    for stage, res in by_stage.items():
        if "error" in res:
            assert isinstance(res["error"], str) and len(res["error"]) > 0
        else:
            assert "coef_table" in res
            assert "r_squared" in res
            assert "n_obs" in res
            has_real_result = True
    assert has_real_result, "At least one stage must produce a real result for page 13 to render"
```
