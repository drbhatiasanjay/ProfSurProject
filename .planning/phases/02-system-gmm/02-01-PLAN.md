# Plan 02-01: System GMM Function Hardening + Unit Tests

## Objective

Replace the methodologically-incorrect OLS stub inside `run_system_gmm` with a proper
`linearmodels.iv.IVGMM` dynamic panel estimator. Preserve the existing return-dict contract
exactly — `pages/13_advanced_econometrics.py` and both test files consume the same keys.

The current implementation uses `statsmodels.OLS` with lag DV as a regressor (Nickell-biased)
and fabricates a pseudo-Hansen J-statistic (`n*(1 - ssr/sst)`) that produces values like 13931
(nonsensical). The replacement uses `IVGMM` with lag2+lag3 as instruments, which provides a
genuine J-statistic via `result.j_stat`.

def run_system_gmm(df, y_col=DEFAULT_Y_COL, x_cols=None, entity="company_code", time="year"):
    """
    System GMM estimation with lagged dependent variable (Arellano-Bond instrument approach).
    Uses linearmodels.iv.IVGMM with lag2 and lag3 of DV as excluded instruments.
    Matches thesis Table 5.12.

    Returns
    -------
    dict with keys:
      type, coef_table, r_squared, adj_r_squared, n_obs, n_firms,
      lag_dv_included (always True), ar1, ar2, sargan, result_obj
    """
    from linearmodels.iv import IVGMM

    if x_cols is None:
        x_cols = DEFAULT_X_COLS

    # Build lag columns for instruments and lag DV regressor
    work = df.sort_values([entity, time]).copy()
    for lag in (1, 2, 3):
        work[f"{y_col}_lag{lag}"] = work.groupby(entity)[y_col].shift(lag)

    # Winsorize y at 1/99 percentile (consistent with prepare_panel convention)
    low, high = work[y_col].quantile(0.01), work[y_col].quantile(0.99)
    work[y_col] = work[y_col].clip(lower=low, upper=high)

    needed = [y_col, f"{y_col}_lag1", f"{y_col}_lag2", f"{y_col}_lag3"] + list(x_cols)
    work = work.dropna(subset=needed).set_index([entity, time])

    if len(work) < 100:
        return {"error": f"Too few observations for GMM ({len(work)}). Need 100+."}

    y     = work[y_col]
    exog  = work[list(x_cols)].copy()
    exog.insert(0, "const", 1.0)
    endog = work[[f"{y_col}_lag1"]]
    instr = work[[f"{y_col}_lag2", f"{y_col}_lag3"]]

    model = IVGMM(y, exog, endog, instr)
    result = model.fit(cov_type="robust")

    coef_table = pd.DataFrame({
        "Variable":    result.params.index.tolist(),
        "Coefficient": result.params.values,
        "Std Error":   result.std_errors.values,
        "t-stat":      result.tstats.values,
        "p-value":     result.pvalues.values,
    })

    # AR(1)/AR(2) tests — Pearson correlation on IVGMM residuals
    resid_df = result.resids.reset_index()
    resid_df.columns = [entity, time, "resid"]
    resid_df = resid_df.sort_values([entity, time])
    resid_df["resid_lag1"] = resid_df.groupby(entity)["resid"].shift(1)
    resid_df["resid_lag2"] = resid_df.groupby(entity)["resid"].shift(2)

    ar1_df = resid_df.dropna(subset=["resid", "resid_lag1"])
    ar2_df = resid_df.dropna(subset=["resid", "resid_lag2"])

    ar1_corr, ar1_p = (
        stats.pearsonr(ar1_df["resid"], ar1_df["resid_lag1"])
        if len(ar1_df) > 10 else (0.0, 1.0)
    )
    ar2_corr, ar2_p = (
        stats.pearsonr(ar2_df["resid"], ar2_df["resid_lag2"])
        if len(ar2_df) > 10 else (0.0, 1.0)
    )

    # Hansen J overidentification test (built into IVGMM result)
    j = result.j_stat  # WaldTestStatistic with .stat, .pval, .df

    return {
        "type": "System GMM",
        "coef_table": coef_table,
        "r_squared": float(result.rsquared),
        "adj_r_squared": float(getattr(result, "rsquared_adj", result.rsquared)),
        "n_obs": int(result.nobs),
        "n_firms": int(work.index.get_level_values(0).nunique()),
        "lag_dv_included": True,
        "ar1": {
            "correlation": float(ar1_corr),
            "p_value":     float(ar1_p),
            "verdict":     ("AR(1) expected significant"
                            if ar1_p < 0.05 else "AR(1) not significant"),
        },
        "ar2": {
            "correlation": float(ar2_corr),
            "p_value":     float(ar2_p),
            "verdict":     ("AR(2) not significant (good)"
                            if ar2_p > 0.05 else
                            "AR(2) significant (instruments may be invalid)"),
        },
        "sargan": {
            "j_stat":  float(j.stat),
            "df":      int(j.df),
            "p_value": float(j.pval),
            "verdict": ("Instruments valid (cannot reject H0)"
                        if j.pval > 0.05 else
                        "Instruments may be invalid (reject H0)"),
        },
        "result_obj": result,
    }
```

def test_system_gmm(self, full_panel):
    """System GMM with lag DV using IVGMM — Arellano-Bond instrument approach."""
    from models.econometric import run_system_gmm
    result = run_system_gmm(full_panel)

    # Structure (unchanged from existing test)
    assert "coef_table" in result
    assert result["lag_dv_included"] is True
    assert "ar1" in result
    assert "ar2" in result
    assert "sargan" in result
    assert result["n_obs"] > 2000

    # coef_table shape and columns
    ct = result["coef_table"]
    assert set(ct.columns) >= {"Variable", "Coefficient", "Std Error", "t-stat", "p-value"}
    assert any("lag" in str(v).lower() for v in ct["Variable"]), (
        "coef_table must contain the lag DV regressor"
    )

    # Lag DV coefficient should be positive and < 1 (capital structure persistence)
    lag_row = ct[ct["Variable"].str.contains("lag", case=False)]
    assert not lag_row.empty
    lag_coef = float(lag_row.iloc[0]["Coefficient"])
    assert 0.0 < lag_coef < 1.0, (
        f"Lag DV coef {lag_coef:.3f} out of [0,1] range — OLS stub may still be active"
    )

    # Hansen J-stat must be a real value (not the old fabricated ~13931)
    assert result["sargan"]["j_stat"] < 1000, (
        f"sargan j_stat={result['sargan']['j_stat']:.1f} — looks like the old OLS pseudo-formula"
    )
    assert 0.0 <= result["sargan"]["p_value"] <= 1.0

    # AR(1) and AR(2) keys
    for ar_key in ("ar1", "ar2"):
        ar = result[ar_key]
        assert "correlation" in ar
        assert "p_value" in ar
        assert "verdict" in ar
        assert -1.0 <= ar["correlation"] <= 1.0
        assert 0.0 <= ar["p_value"] <= 1.0

    # type must not claim OLS
    assert "OLS" not in result.get("type", ""), (
        f"type='{result.get('type')}' still mentions OLS — IVGMM not active"
    )
```
