```python
def test_system_gmm(self, thesis_panel):
    """Page 13 GMM tab: run_system_gmm returns full valid dict on thesis panel."""
    from models.econometric import run_system_gmm

    r = run_system_gmm(thesis_panel)

    # Must not error on the thesis panel
    assert "error" not in r, f"run_system_gmm errored: {r.get('error')}"

    # All keys that page 13 reads (from pages/13_advanced_econometrics.py lines 95-148)
    required_keys = ["r_squared", "n_obs", "n_firms", "coef_table", "ar1", "ar2", "sargan"]
    for key in required_keys:
        assert key in r, f"Missing key consumed by page 13: '{key}'"

    # coef_table columns page 13 passes to format_coef_table
    ct = r["coef_table"]
    assert set(ct.columns) >= {"Variable", "Coefficient", "Std Error", "t-stat", "p-value"}

    # AR dicts — page 13 reads ["correlation"], ["p_value"], ["verdict"]
    for ar_key in ("ar1", "ar2"):
        ar = r[ar_key]
        assert "correlation" in ar
        assert "p_value"     in ar
        assert "verdict"     in ar

    # Sargan dict — page 13 reads ["j_stat"], ["p_value"], ["verdict"]
    s = r["sargan"]
    assert "j_stat"  in s
    assert "p_value" in s
    assert "verdict" in s

    # Numeric sanity
    assert r["r_squared"] > 0.0
    assert r["n_obs"] > 0
    assert r["n_firms"] > 0
    assert 0.0 <= s["p_value"] <= 1.0

    # Not the old OLS stub
    assert s["j_stat"] < 1000, (
        f"j_stat={s['j_stat']:.1f} — OLS pseudo-formula detected; IVGMM not active"
    )
```
