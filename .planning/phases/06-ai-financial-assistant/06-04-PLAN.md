"""Page 19: AI Financial Assistant — full-screen dedicated chat.

Shares st.session_state['chat_history'] with the floating bubble in app.py,
so a user can switch between modalities without losing context.

Roles: admin, researcher (viewer is denied).
"""
import json
import streamlit as st

import db
from helpers import require_role, plotly_layout
from models.llm_adapters import (
    build_company_context,
    build_panel_context,
    stream_ollama,
    stream_anthropic,
    parse_llm_json,
    log_chat_query,
    count_tokens,
    classify_query,
)

# ---- Role gate ----
require_role(["admin", "researcher"])

st.title("AI Financial Assistant")
st.caption(
    "Ask grounded questions about any company in the panel or about panel-wide "
    "patterns. Replies are constrained to the data in your current context — "
    "no hallucinated numbers."
)


# ---- Helpers ----
JSON_SCHEMA_INSTRUCTION = (
    "\n\nReply STRICTLY as a JSON object with keys: "
    '{"answer": str, "citations": [str], "followup_questions": [str], '
    '"chart_request": null | {"type": str, "x": str, "y": str}}'
)


def suggested_questions(mode: str) -> list[str]:
    """Return 3-5 starter prompts tuned to the current mode."""
    if mode == "cfo":
        return [
            "How does our leverage compare to peers in our industry and stage?",
            "Which determinants of leverage matter most for a company like ours?",
            "What is the 5-year leverage trend for this company?",
            "Recommend three capital-structure actions based on the panel evidence.",
        ]
    return [
        "What is the mean leverage across the panel by life stage?",
        "Which OLS regressors are statistically significant at p<0.05?",
        "How does leverage differ between Birth and Mature firms?",
        "Summarise the key findings from the baseline OLS model.",
    ]


def init_state():
    if "chat_history" not in st.session_state:
        st.session_state["chat_history"] = []
    if "chat_mode" not in st.session_state:
        st.session_state["chat_mode"] = "researcher"
    if "chat_backend" not in st.session_state:
        st.session_state["chat_backend"] = "ollama"
    if "chat_json_mode" not in st.session_state:
        st.session_state["chat_json_mode"] = False


init_state()


# ---- Sidebar controls ----
with st.sidebar:
    st.markdown("### Assistant Settings")
    st.session_state["chat_mode"] = st.radio(
        "Mode",
        ["researcher", "cfo"],
        index=["researcher", "cfo"].index(st.session_state["chat_mode"]),
        format_func=lambda x: {"researcher": "Researcher (panel-wide)",
                               "cfo": "CFO (single company)"}[x],
        help="Researcher injects panel OLS + summary stats. CFO injects "
             "company KPIs + peer benchmarks.",
    )
    st.session_state["chat_backend"] = st.radio(
        "Backend",
        ["ollama", "anthropic"],
        index=["ollama", "anthropic"].index(st.session_state["chat_backend"]),
        format_func=lambda x: {"ollama": "Ollama (local)",
                               "anthropic": "Claude (cloud)"}[x],
        help="Ollama runs locally with zero data egress. Anthropic sends your "
             "context to Claude API.",
    )
    if st.session_state["chat_backend"] == "anthropic":
        st.warning("Your context + question will leave the server.")
    st.session_state["chat_json_mode"] = st.checkbox(
        "Structured JSON output",
        value=st.session_state["chat_json_mode"],
        help="Reply as parsed card (answer + citations + follow-ups).",
    )
    if st.session_state["chat_mode"] == "cfo":
        cur = st.session_state.get("active_company_cin")
        if cur:
            st.info(f"Active CIN: {cur}")
        else:
            cin_input = st.text_input("CIN", help="Set the company for CFO mode")
            if cin_input:
                st.session_state["active_company_cin"] = cin_input.strip()
                st.rerun()
    if st.button("Clear chat history"):
        st.session_state["chat_history"] = []
        st.rerun()


# ---- Suggested questions ----
mode = st.session_state["chat_mode"]
if not st.session_state["chat_history"]:
    st.markdown("**Try one of these to start:**")
    cols = st.columns(2)
    for i, q in enumerate(suggested_questions(mode)):
        col = cols[i % 2]
        if col.button(q, key=f"suggest_{i}"):
            # Inject as a pending question — handled below
            st.session_state["__pending_q"] = q
            st.rerun()
```

# ---- Chat history render ----
for turn in st.session_state["chat_history"]:
    with st.chat_message(turn["role"]):
        # If JSON-mode card payload is stashed, render structured
        if turn.get("structured"):
            s = turn["structured"]
            st.markdown(s.get("answer", ""))
            if s.get("citations"):
                with st.expander("Citations"):
                    for c in s["citations"]:
                        st.markdown(f"- {c}")
            if s.get("followup_questions"):
                st.markdown("**Follow-up questions:**")
                for q in s["followup_questions"]:
                    st.markdown(f"- {q}")
        else:
            st.markdown(turn["content"])


# ---- Send loop ----
pending = st.session_state.pop("__pending_q", None)
typed = st.chat_input("Ask anything about your data...")
user_q = pending or typed

if user_q:
    backend = st.session_state["chat_backend"]
    mode = st.session_state["chat_mode"]
    json_mode = st.session_state["chat_json_mode"]

    # Build context
    if mode == "cfo" and st.session_state.get("active_company_cin"):
        ctx = build_company_context(
            st.session_state["active_company_cin"],
            panel_mode=st.session_state.get("panel_mode", "thesis"),
        )
    else:
        ctx = build_panel_context(
            panel_mode=st.session_state.get("panel_mode", "thesis"),
        )
    if json_mode:
        ctx = ctx + JSON_SCHEMA_INSTRUCTION

    # Append user turn + render
    st.session_state["chat_history"].append({"role": "user", "content": user_q})
    with st.chat_message("user"):
        st.markdown(user_q)

    # Build messages list (last 5 turns + current)
    history_msgs = [
        {"role": t["role"], "content": t.get("content") or t.get("structured", {}).get("answer", "")}
        for t in st.session_state["chat_history"][-11:-1]
    ]
    messages = history_msgs + [{"role": "user", "content": user_q}]

    # Stream reply
    with st.chat_message("assistant"):
        if json_mode:
            # Buffer entire reply, parse, then render — never stream raw JSON
            with st.spinner("Thinking..."):
                if backend == "ollama":
                    chunks = list(stream_ollama(
                        [{"role": "system", "content": ctx}] + messages
                    ))
                else:
                    chunks = list(stream_anthropic(messages, system=ctx))
                full = "".join(chunks)
                parsed = parse_llm_json(full)
            st.markdown(parsed.get("answer", ""))
            if parsed.get("citations"):
                with st.expander("Citations"):
                    for c in parsed["citations"]:
                        st.markdown(f"- {c}")
            if parsed.get("followup_questions"):
                st.markdown("**Follow-up questions:**")
                for q in parsed["followup_questions"]:
                    st.markdown(f"- {q}")
            st.session_state["chat_history"].append({
                "role": "assistant",
                "content": full,
                "structured": parsed,
            })
        else:
            if backend == "ollama":
                full = st.write_stream(stream_ollama(
                    [{"role": "system", "content": ctx}] + messages
                ))
            else:
                full = st.write_stream(stream_anthropic(messages, system=ctx))
            st.session_state["chat_history"].append({
                "role": "assistant", "content": full or "",
            })

    # Audit log — distinct page_name to separate from bubble queries
    user = st.session_state.get("user", {}) or {}
    try:
        details = json.dumps({
            "llm_backend": backend,
            "token_count": count_tokens(ctx) + count_tokens(user_q) + count_tokens(full or ""),
            "query_preview": (user_q or "")[:200],
            "mode": mode,
            "json_mode": json_mode,
            "classification": classify_query(user_q),
        })
        # Use db.get_connection() — the actual API (db._connection does not exist)
        conn = db.get_connection()
        try:
            conn.execute(
                "INSERT INTO audit_log(username, role, page_name, action_type, details, session_id) "
                "VALUES (?,?,?,?,?,?)",
                (user.get("username", "anonymous"),
                 user.get("role", "viewer"),
                 "ai_assistant_page",
                 "ai_query",
                 details,
                 st.session_state.get("session_id", "")),
            )
            conn.commit()
        finally:
            try:
                conn.close()
            except Exception:
                pass
    except Exception:
        pass


# ---- Footer ----
st.markdown("---")
st.caption(
    f"Backend: **{st.session_state['chat_backend']}** | "
    f"Mode: **{st.session_state['chat_mode']}** | "
    f"Turns: **{len(st.session_state['chat_history'])}** | "
    "Shared history with the floating bubble. "
    "[Activity Log →](/admin_activity)"
)
```
